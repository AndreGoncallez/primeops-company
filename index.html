function handleFormSubmit(event) { // Previne a navegação imediata do formulário tradicional event.preventDefault(); const form = event.target; const lang = document.getElementById('html-root').lang || 'en'; const name = document.getElementById('input-name').value; const email = document.getElementById('input-email').value; const msg = document.getElementById('modal-msg').value; // Coleta serviços marcados nas checkboxes const selectedServices = []; document.querySelectorAll('input[name="scope[]"]:checked').forEach(cb => { selectedServices.push(cb.value); }); const servicesList = selectedServices.length > 0 ? selectedServices.join(', ') : (lang === 'pt' ? 'Nenhum específico' : 'None specified'); // Dispara o envio em segundo plano para o Formspree (para ficar salvo no e-mail/dashboard) fetch(form.action, { method: 'POST', body: new FormData(form), headers: { 'Accept': 'application/json' } }).catch(err => console.error('Formspree dispatch error:', err)); // Formatação da mensagem direta para o WhatsApp ou SMS const phoneNumber = "5585981326868"; let fullMessage = ""; if (lang === 'pt') { fullMessage = `*Nova Solicitação - PrimeOps Site*\n\n` + `👤 *Nome/Empresa:* ${name}\n` + `📧 *E-mail:* ${email}\n` + `🛠️ *Serviços Selecionados:* ${servicesList}\n` + `📝 *Detalhes:* ${msg}`; // Abre o WhatsApp com a mensagem formatada const waUrl = `https://wa.me/${phoneNumber}?text=${encodeURIComponent(fullMessage)}`; window.open(waUrl, '_blank'); } else { fullMessage = `New PrimeOps Inquiry:\n` + `Name: ${name}\n` + `Email: ${email}\n` + `Services: ${servicesList}\n` + `Details: ${msg}`; // Abre o aplicativo de SMS com a mensagem formatada const smsUrl = `sms:+${phoneNumber}&body=${encodeURIComponent(fullMessage)}`; window.location.href = smsUrl; } closeModal(); }
